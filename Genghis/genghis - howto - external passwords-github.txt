using System.Text.Json;

var path = Path.Combine(AppContext.BaseDirectory, "radar-passwords.json");

if (!File.Exists(path))
    throw new FileNotFoundException(
        "Create radar-passwords.json from radar-passwords.example.json.",
        path);

var allPasswords =
    JsonSerializer.Deserialize<Dictionary<string, Dictionary<int, string>>>(
        File.ReadAllText(path))
    ?? throw new InvalidOperationException("Password file is empty or invalid.");

string section =
    ON_TESTSERVER ? "TestServer" :
    ON_JUBILEE ? "Jubilee" :
    "Default";

if (!allPasswords.TryGetValue(section, out var radarpasswords))
    throw new InvalidOperationException(
        $"Password section '{section}' is missing.");
